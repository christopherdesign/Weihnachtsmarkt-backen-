function imageFromFile(file){
 return new Promise((resolve,reject)=>{
   const reader=new FileReader();
   reader.onerror=()=>reject(new Error('Foto konnte nicht gelesen werden.'));
   reader.onload=()=>{
     const img=new Image();img.onerror=()=>reject(new Error('Das ausgewählte Bild ist ungültig.'));
     img.onload=()=>resolve(img);img.src=reader.result;
   };
   reader.readAsDataURL(file);
 });
}
async function compressReceiptPhoto(file){
 if(!file?.type?.startsWith('image/'))throw new Error('Bitte ein Bild als Bon-Foto auswählen.');
 if(file.size>15*1024*1024)throw new Error('Das Foto ist zu groß. Bitte ein Bild unter 15 MB auswählen.');
 const img=await imageFromFile(file),maxSide=1400;
 let scale=Math.min(1,maxSide/Math.max(img.naturalWidth,img.naturalHeight));
 let width=Math.max(1,Math.round(img.naturalWidth*scale)),height=Math.max(1,Math.round(img.naturalHeight*scale));
 let quality=.78,dataUrl='';
 for(let attempt=0;attempt<9;attempt++){
   const canvas=document.createElement('canvas');canvas.width=width;canvas.height=height;
   const ctx=canvas.getContext('2d',{alpha:false});ctx.fillStyle='#fff';ctx.fillRect(0,0,width,height);ctx.drawImage(img,0,0,width,height);
   dataUrl=canvas.toDataURL('image/jpeg',quality);
   if(dataUrl.length<=180000)break;
   if(quality>.48)quality-=.08;else{width=Math.max(500,Math.round(width*.82));height=Math.max(1,Math.round(img.naturalHeight*(width/img.naturalWidth)));}
 }
 if(dataUrl.length>260000)throw new Error('Das Foto konnte nicht ausreichend verkleinert werden. Bitte den Bon enger fotografieren.');
 return {dataUrl,name:file.name||'Bon-Foto.jpg',originalSize:file.size};
}
