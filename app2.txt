function metricsForState(state){
 if(!state)return {revenue:0,shop:0,energy:0,profit:0,buckets:0,cookie:0,horn:0,mixed:0};
 const entries=Array.isArray(state.saleEntries)?state.saleEntries:[];
 const sales=entries.reduce((a,e)=>{
   a.cookie+=n(e.cookieQty);a.horn+=n(e.hornQty);a.mixed+=n(e.mixedQty);
   a.revenue+=n(e.cookieQty)*n(e.cookiePrice)+n(e.hornQty)*n(e.hornPrice)+n(e.mixedQty)*n(e.mixedPrice);
   return a;
 },{cookie:0,horn:0,mixed:0,revenue:0});
 const shop=(state.shopping||[]).reduce((a,x)=>a+(Math.max(0,Math.round(n(x.qty)))*n(x.unitPrice!==undefined?x.unitPrice:(n(x.qty)>0?n(x.cost)/n(x.qty):0))),0);
 const timerElapsed=d=>(d?.elapsed||0)+(d?.running&&d?.startAt?Date.now()-d.startAt:0);
 const ovenKwh=(n(state.oven?.watts)/1000)*(timerElapsed(state.oven)/3600000);
 const ironKwh=((n(state.iron?.count)||1)*(n(state.iron?.wattsEach)||800)/1000)*(timerElapsed(state.iron)/3600000);
 const logs=(state.energyLogs||[]).reduce((a,l)=>{
   if(!l.start||!l.end)return a;
   const [sh,sm]=l.start.split(':').map(Number),[eh,em]=l.end.split(':').map(Number);
   let mins=(eh*60+em)-(sh*60+sm);if(mins<0)mins+=1440;
   const watts=l.device==='iron'?(n(l.ironCount)||1)*(n(state.iron?.wattsEach)||800):n(state.oven?.watts);
   return a+(watts/1000)*(mins/60);
 },0);
 const energy=(ovenKwh+ironKwh+logs)*n(state.powerPrice);
 return {revenue:sales.revenue,shop,energy,profit:sales.revenue-shop-energy,buckets:sales.cookie+sales.horn+sales.mixed,cookie:sales.cookie,horn:sales.horn,mixed:sales.mixed};
}
function deltaInfo(cur,prev,invert=false){
 if(!prev && prev!==0)return {text:'–',cls:'flat'};
 if(prev===0){
   if(cur===0)return {text:'± 0 %',cls:'flat'};
   return {text:'neu',cls:invert?'down':'up'};
 }
 const pct=((cur-prev)/Math.abs(prev))*100;
 const good=invert?pct<=0:pct>=0;
 return {text:(pct>=0?'+':'')+pct.toLocaleString('de-DE',{maximumFractionDigits:1})+' %',cls:Math.abs(pct)<.05?'flat':good?'up':'down'};
}
function applyDelta(id,cur,prev,invert=false){
 const el=$(id),d=deltaInfo(cur,prev,invert);if(!el)return;el.textContent=d.text;el.className='delta '+d.cls;
}
