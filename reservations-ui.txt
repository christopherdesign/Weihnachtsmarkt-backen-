     const record=readYearRecord(year), conflict=record.conflict;
     if(!conflict)continue;
     const title=document.createElement('h3');title.textContent=String(year);list.append(title);
     const hint=document.createElement('p');
     hint.textContent=[conflict.cause,stockConflictHelp[conflict.code],'Die lokalen Änderungen dieses Jahres bleiben gespeichert; die Übertragung pausiert.'].filter(Boolean).join(' ');
     list.append(hint);
     for(const id of conflict.ids){
       const info=record.details?.[id],p=document.createElement('p');
       const actions={create:'Reservierung angelegt',edit:'Reservierung bearbeitet',change:'Menge geändert',pickup:'Abholung gebucht',cancel:'Reservierung storniert',reverse:'Abholung korrigiert'};
       p.textContent=[info?.at&&new Date(info.at).toLocaleString('de-DE'),
         info?.operation&&actions[info.operation.type],info?.customer,
         ...(info?.targets||[]).map(t=>t.label+' '+t.id),id].filter(Boolean).join(' · ');
       list.append(p);
     }
     const retry=document.createElement('button');retry.type='button';retry.className='secondary';
     retry.textContent='Nach Prüfung erneut versuchen';
     retry.disabled=!IS_PRODUCTION_DEPLOYMENT||!navigator.onLine;
     retry.onclick=()=>{requestedSyncRetries.add(year);initCloud(true)};
     list.append(retry);
   }
   conflicts.hidden=!list.childElementCount;
 };
