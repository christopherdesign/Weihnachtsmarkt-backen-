 if(!SYNC_PATHS.includes('energyConfig'))SYNC_PATHS.push('energyConfig');
 function cfg(){s.energyConfig=s.energyConfig||{prices:{},devices:[]};s.energyConfig.prices=s.energyConfig.prices||{};s.energyConfig.devices=s.energyConfig.devices||[];return s.energyConfig}
 function famPrice(f){return n(cfg().prices[f])||n(s.powerPrice)||.25}
 function icon(t){return t==='Backofen'?'🔥':t==='Hörncheneisen'?'🧇':t==='Küchenmaschine'?'🥣':t==='Herd'?'🍳':'⚡'}
 function esc(v){return String(v??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
 function ironCount(d,value){return Math.min(12,Math.max(1,Math.round(n(value??d.ironCount??s.iron?.count)||1)))}
 function effectiveWatts(d,count){return n(d.watts)*(d.type==='Hörncheneisen'?ironCount(d,count):1)}
 function devElapsed(d){return d.running&&d.startAt?Math.max(0,Date.now()-d.startAt):0}
 function devCost(d){return effectiveWatts(d)/1000*(devElapsed(d)/3600000)*famPrice(d.family)}
 function energyByFamily(){
   const out={[s.family1]:0,[s.family2]:0};
   (s.energyLogs||[]).forEach(l=>{const f=l.family||'';if(Object.hasOwn(out,f))out[f]+=energyLogCost(l)});
   cfg().devices.forEach(d=>{if(Object.hasOwn(out,d.family))out[d.family]+=devCost(d)});
   return out;
 }
