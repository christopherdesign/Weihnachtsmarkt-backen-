async function syncYear(year,retryConflict=false){

   const stored=readYearRecord(year);
   const local=stored?.state||archivedYear(year)||(Number(s.year)===year?lastSavedSnapshot:null);
   const pending=stored?.pending||[];
   if(stored?.conflict&&!retryConflict)return 'conflict';

   const visibleStamp=await getCloudStamp(year);
   if(!pending.length&&cloudStamps.has(year)&&cloudStamps.get(year)===visibleStamp)return;
   const row=visibleStamp?await getCloudRow(year):null;
   if(visibleStamp&&!row)throw new Error('Cloud-Jahr beim Lesen verschwunden');
   if(!row&&(!local||isPristineLocalState(local))) {cloudStamps.set(year,'');return}
   const remote=row?normalizeState(row.data,year,row.data?._sync?.migratedAt||row.updated_at):null;
   let merged;
   try{merged=cloudCandidate(local,remote,row,year,operations)}
   catch(error){
     if(!error.reservationOperation)throw error;
     holdSyncConflict(year,'WM_STOCK_STALE_RESERVATION',[error.reservationOperation],error.message);
     return 'conflict';
   }
   let acknowledged=row?.updated_at||'';
   if(!remote||!same(merged,remote)){
     const stamp=new Date(Math.max(Date.now(),(Date.parse(acknowledged)||0)+1)).toISOString();
     const query='app_state?select=updated_at';
     const options={headers:{Prefer:'return=representation'},body:JSON.stringify(row?{data:merged,updated_at:stamp}:{workspace_id:workspaceId(),year,data:merged,updated_at:stamp})};
     let saved;
     try{
       saved=await cloudRequest(row?query+'&workspace_id=eq.'+encodeURIComponent(workspaceId())+'&year=eq.'+year+'&updated_at=eq.'+encodeURIComponent(acknowledged):query,{...options,method:row?'PATCH':'POST'});
     }catch(error){
       let code;try{code=JSON.parse(error.detail).message}catch{}
       if(Object.hasOwn(stockConflictHelp,code)){holdSyncConflict(year,code,pending);return 'conflict'}
       if(error.status===409)continue;
       throw error;
     }
     if(!saved?.[0]?.updated_at)continue;
     acknowledged=saved[0].updated_at;
   }
   const latest=readYearRecord(year);
   const remaining=(latest?.pending||[]).filter(id=>!pending.includes(id));
   const concurrent=remaining.length>0;
   let next=merged, conflict=null;
   if(concurrent){
     try{next=cloudCandidate(latest.state,merged,{data:merged,updated_at:acknowledged},year,remaining.map(id=>latest.details?.[id]?.operation).filter(Boolean))}
     catch(error){
       next=latest.state;
       conflict={code:'WM_STOCK_STALE_RESERVATION',ids:[error.reservationOperation],cause:error.message,at:nowIso()};
     }
   }
   if(concurrent&&latest.state._sync?.lastLocalChange)next._sync.lastLocalChange=latest.state._sync.lastLocalChange;

   if(Number(s.year)===year){s=next;lastSavedSnapshot=deep(s);render()}
   cloudStamps.set(year,acknowledged);
   return;
 }
 throw new Error('Cloud-Konflikt konnte nicht aufgelöst werden');
}
